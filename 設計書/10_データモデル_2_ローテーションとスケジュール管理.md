# 練習表自動生成システム 設計書：データモデル（2/3）- ローテーションとスケジュール管理

## 3. ローテーションとスケジュール管理エンティティ

ローテーションとスケジュール管理に関連するエンティティの詳細設計を以下に示します。

### 3.1 スケジュール割当エンティティ

#### 3.1.1 SupervisorAssignment（監督者割当）

セッションに対する監督者の割当を表すエンティティです。

| 属性 | 型 | 説明 | 制約 |
|-----|-----|-----|-----|
| assignment_id | UUID | 割当の一意識別子 | PK, NOT NULL |
| session_id | UUID | セッションID | FK (PracticeSession), NOT NULL |
| supervisor_id | UUID | 監督者ID | FK (Member), NOT NULL |
| assignment_type | ENUM | 割当タイプ | DEFAULT 'PRIMARY' |
| status | ENUM | 状態 | DEFAULT 'ASSIGNED' |
| notes | TEXT | 備考 | |
| assigned_by | UUID | 割当者ID | FK (Member), NOT NULL |
| assigned_at | TIMESTAMP | 割当日時 | DEFAULT CURRENT_TIMESTAMP |
| updated_at | TIMESTAMP | 更新日時 | DEFAULT CURRENT_TIMESTAMP ON UPDATE |

```sql
CREATE TABLE supervisor_assignments (
    assignment_id UUID PRIMARY KEY,
    session_id UUID NOT NULL,
    supervisor_id UUID NOT NULL,
    assignment_type ENUM('PRIMARY', 'BACKUP', 'ASSISTANT') DEFAULT 'PRIMARY',
    status ENUM('ASSIGNED', 'CONFIRMED', 'DECLINED', 'COMPLETED') DEFAULT 'ASSIGNED',
    notes TEXT,
    assigned_by UUID NOT NULL,
    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (session_id) REFERENCES practice_sessions(session_id) ON DELETE CASCADE,
    FOREIGN KEY (supervisor_id) REFERENCES members(member_id),
    FOREIGN KEY (assigned_by) REFERENCES members(member_id),
    UNIQUE (session_id, supervisor_id)
);
```

### 3.2 可用性管理エンティティ

#### 3.2.1 MemberAvailability（メンバー可用性）

メンバーの練習参加可能な時間帯を表すエンティティです。

| 属性 | 型 | 説明 | 制約 |
|-----|-----|-----|-----|
| availability_id | UUID | 可用性の一意識別子 | PK, NOT NULL |
| member_id | UUID | メンバーID | FK (Member), NOT NULL |
| day_of_week | ENUM | 曜日 | NOT NULL |
| start_time | TIME | 開始時間 | NOT NULL |
| end_time | TIME | 終了時間 | NOT NULL |
| preference_level | ENUM | 希望強度 | DEFAULT 'PREFERRED' |
| recurrence_type | ENUM | 繰り返しタイプ | DEFAULT 'WEEKLY' |
| effective_from | DATE | 有効開始日 | DEFAULT CURRENT_DATE |
| effective_until | DATE | 有効終了日 | NULL |
| created_at | TIMESTAMP | 作成日時 | DEFAULT CURRENT_TIMESTAMP |
| updated_at | TIMESTAMP | 更新日時 | DEFAULT CURRENT_TIMESTAMP ON UPDATE |

```sql
CREATE TABLE member_availabilities (
    availability_id UUID PRIMARY KEY,
    member_id UUID NOT NULL,
    day_of_week ENUM('MONDAY', 'TUESDAY', 'WEDNESDAY', 'THURSDAY', 'FRIDAY', 'SATURDAY', 'SUNDAY') NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    preference_level ENUM('UNAVAILABLE', 'AVAILABLE', 'PREFERRED', 'HIGHLY_PREFERRED') DEFAULT 'PREFERRED',
    recurrence_type ENUM('WEEKLY', 'BIWEEKLY', 'MONTHLY', 'ONCE') DEFAULT 'WEEKLY',
    effective_from DATE DEFAULT CURRENT_DATE,
    effective_until DATE NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (member_id) REFERENCES members(member_id) ON DELETE CASCADE,
    CHECK (end_time > start_time),
    CHECK (effective_until IS NULL OR effective_until > effective_from)
);
```

#### 3.2.3 AbsenceRequest（欠席申請）

| 属性 | 型 | 説明 | 制約 |

| reason | TEXT | 理由 | |
| status | ENUM | 状態 | DEFAULT 'PENDING' |

```sql
CREATE TABLE absence_requests (
    absence_id UUID PRIMARY KEY,
    member_id UUID NOT NULL,
    session_id UUID NULL,
    date DATE NOT NULL,
    reason TEXT,
    status ENUM('PENDING', 'APPROVED', 'REJECTED', 'CANCELLED') DEFAULT 'PENDING',
    is_recurring BOOLEAN DEFAULT FALSE,
    recurrence_end_date DATE NULL,

    FOREIGN KEY (session_id) REFERENCES practice_sessions(session_id) ON DELETE SET NULL,

    CHECK (recurrence_end_date IS NULL OR recurrence_end_date >= date)

```

```sql
CREATE TABLE schedule_generations (
    generation_id UUID PRIMARY KEY,
    schedule_id UUID NOT NULL,
    generation_parameters JSON NOT NULL,
    algorithm_version VARCHAR(20) NOT NULL,
    status ENUM('PENDING', 'IN_PROGRESS', 'COMPLETED', 'FAILED', 'CANCELLED') DEFAULT 'COMPLETED',
    result_metrics JSON,
    executed_by UUID NOT NULL,

    error_message TEXT NULL,
    FOREIGN KEY (schedule_id) REFERENCES schedules(schedule_id),
    FOREIGN KEY (executed_by) REFERENCES members(member_id)

```

| 属性 | 型 | 説明 | 制約 |

```sql
CREATE TABLE optimization_parameters (
```

### 4.2 評価とフィードバック

#### 4.2.1 ScheduleEvaluation（スケジュール評価）

| 属性 | 型 | 説明 | 制約 |
|-----|-----|-----|-----|
| evaluation_id | UUID | 評価の一意識別子 | PK, NOT NULL |
| schedule_id | UUID | スケジュールID | FK (Schedule), NOT NULL |
| evaluator_id | UUID | 評価者ID | FK (Member), NOT NULL |
| overall_rating | INTEGER | 総合評価（1-5） | NOT NULL |
| fairness_rating | INTEGER | 公平性評価（1-5） | |
| efficiency_rating | INTEGER | 効率性評価（1-5） | |
| convenience_rating | INTEGER | 利便性評価（1-5） | |
| comments | TEXT | コメント | |
| evaluated_at | TIMESTAMP | 評価日時 | DEFAULT CURRENT_TIMESTAMP |
| updated_at | TIMESTAMP | 更新日時 | DEFAULT CURRENT_TIMESTAMP ON UPDATE |

```sql
CREATE TABLE schedule_evaluations (
    evaluation_id UUID PRIMARY KEY,
    schedule_id UUID NOT NULL,
    evaluator_id UUID NOT NULL,
    overall_rating INTEGER NOT NULL,
    fairness_rating INTEGER,
    efficiency_rating INTEGER,
    convenience_rating INTEGER,
    comments TEXT,

    CHECK (fairness_rating IS NULL OR fairness_rating BETWEEN 1 AND 5),

```
